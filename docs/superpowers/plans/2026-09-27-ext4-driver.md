# 统一 ext2/3/4 驱动（ext4 + 元数据日志）实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 subagent-driven-development（推荐）或 executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 为 NTCLKS 存储栈实现统一 ext 族驱动（ext4 磁盘布局 + JBD2 元数据日志 + 崩溃恢复），发布镜像与安装器根分区默认切到 ext4+日志，ext2/ext3 旧卷保持兼容挂载。

**架构：** 演进式统一驱动（方案 A）：现有 `storage_ext2*.c` 保留为共用元数据核心，新增特性识别层（`storage_ext4_super.c`）、extent 映射层（`storage_ext4_extent.c`）与独立日志组件（`storage_journal.c`）；挂载时按特性分派，`has_journal` 卷先恢复再放行 rw。内核改动落在 NTCLKS 子仓，父仓改工具链/安装器/文档并更新 gitlink。

**技术栈：** C11（NTCLKS 内核存储栈，单 TU：`storage.c` umbrella include）、Python3（镜像工具）、宿主机 e2fsprogs（mke2fs/e2fsck）、QEMU 集成测试。

**规格：** `docs/superpowers/specs/2026-09-27-ext4-driver-design.md`（计划的论证依据来自规格，执行者两份都读）

创建 `storage_ext4_super.c`：

```c
/* ext 族特性识别与安全阀。超级块缓冲由调用方给出（偏移 1024 起 1024 字节）。 */
static int extfs_probe(const uint8_t *super_bytes, struct extfs_features *out)
{
    const struct ext2_superblock *sb = (const struct ext2_superblock *)(const void *)super_bytes;
    memset(out, 0, sizeof(*out));
    if (sb->magic != EXT2_SUPER_MAGIC) return -22; /* -EINVAL */
    out->compat = sb->feature_compat;
    out->incompat = sb->feature_incompat;
    out->ro_compat = sb->feature_ro_compat;
    /* 安全阀：未识别 incompat 位一律拒绝。 */
    if (out->incompat & ~EXT4_FEATURE_INCOMPAT_SUPPORTED) return -95; /* -EOPNOTSUPP */
    out->ro_compat_unknown = (out->ro_compat & ~EXT4_FEATURE_RO_COMPAT_SUPPORTED) != 0;
    out->has_journal = (out->incompat & 0x0004u) != 0; /* INCOMPAT_JOURNAL_DEV 之上的 has_journal 由 compat 0x4 表示 */
    out->has_journal = (sb->feature_compat & 0x0004u) != 0; /* COMPAT_HAS_JOURNAL */
    out->has_extents = (out->incompat & EXT4_FEATURE_INCOMPAT_EXTENTS) != 0;
    out->has_64bit = (out->incompat & EXT4_FEATURE_INCOMPAT_64BIT) != 0;
    out->has_flex_bg = (out->incompat & EXT4_FEATURE_INCOMPAT_FLEX_BG) != 0;
    out->has_metadata_csum = (out->ro_compat & EXT4_FEATURE_RO_COMPAT_METADATA_CSUM) != 0;
    out->inode_size = sb->rev_level >= 1 && sb->inode_size ? sb->inode_size : 128;
    if (out->inode_size < 128 || (out->inode_size & (out->inode_size - 1))) return -22;
    out->desc_size = out->has_64bit ? 64 : 32;
    out->first_ino = sb->rev_level >= 1 && sb->first_ino ? sb->first_ino : 11;
    out->log_groups_per_flex = out->has_flex_bg ? super_bytes[0x174] : 0;
    memcpy(&out->checksum_seed, super_bytes + 0x270, 4);
    if (out->has_journal) out->flavor = out->has_extents ? EXTFS_FLAVOR_EXT4 : EXTFS_FLAVOR_EXT3;
    else out->flavor = EXTFS_FLAVOR_EXT2;
    return 0;
}
```

（注意：`has_journal` 的来源是 compat 位 `0x4`（HAS_JOURNAL），不是 incompat；上面第一行赋值是错误示例已被第二行覆盖——实现时只保留 compat 判定这**一行**，不要提交两行。）

`storage_ext4_super.c` 追加实现：

```c
static const uint32_t extfs_crc32c_table[256] = { /* 标准 CRC32C (Castagnoli) 表，生成式填充：
    0x00000000,0xF26B8303,0xE13B70F7,0x1350F3F4,0xC79A971F,0x35F1141C,0x26A1E7E8,0xD4CA64EB, ... 余 248 项按反射多项式 0x82F63B78 递推 */ };

uint32_t extfs_crc32c(uint32_t seed, const uint8_t *data, size_t len)
{
    uint32_t crc = seed;
    for (size_t i = 0; i < len; ++i)
        crc = extfs_crc32c_table[(crc ^ data[i]) & 0xFF] ^ (crc >> 8);
    return crc;
}
```

（CRC32C 表是纯机械展开：对 i=0..255 递推 `c = i; for (k=0;k<8;k++) c = (c & 1) ? 0x82F63B78u ^ (c >> 1) : c >> 1; table[i] = c;`。提交前用该递推在测试里对拍 `extfs_crc32c` 的已知向量，表体可由同递推一次性展开成字面量——两种写法都要保证向量过。）

```c
/* 定位第 group 个组描述符：GDT 起始块 = first_data_block+1（4K 块时为 1），flex_bg 不影响
 * 描述符表本身的线性布局，仅影响位图/inode 表的放置（那里用 hi 字段拼 64 位物理块号）。 */
int extfs_group_desc(const char *image_path, const struct extfs_features *f,
                     uint32_t group, uint8_t out[64])
{
    FILE *file = fopen(image_path, "rb");
    if (!file) return -2;
    uint32_t block_size = 1024u << 0; /* 由 probe 结果传入的 log_block_size 换算；调用方镜像均 ≥1K */
    long gdt_offset = (long)(f->desc_size ? f->desc_size : 32) * (long)group;
    /* 1K 块：GDT 从块 2 开始；4K 块：从块 1 开始。调用方传入镜像路径时块大小由 extfs_probe 的
     * 调用者写入 f->desc_size 所在会话状态；此函数用固定偏移 2048 起步（1K 块 GDT=块2），
     * 对 4K 块镜像 GDT 在偏移 4096，测试镜像统一 4K，偏移由 gdt_block 参数化： */
    (void)block_size; (void)gdt_offset;
    return -95;
```

```c
/* 将 inode body（60 字节 i_block 区）里的 extent 树 logical 映射到物理块。
 * 返回 0=命中并写 *physical；1=空洞；-5=-EIO 树损坏；-22=-EINVAL 魔数错。 */
static int ext4_extent_walk(const struct extfs_features *f,
                            const uint8_t *node, uint64_t node_phys,
                            uint32_t logical, uint32_t *physical, int depth_left)
{
    const struct ext4_extent_header *eh = (const struct ext4_extent_header *)(const void *)node;
    if (eh->eh_magic != EXT4_EXTENT_MAGIC) return -22;
    if (depth_left < 0 || eh->eh_depth > 3) return -5;
    if (eh->eh_depth == 0) {
        const struct ext4_extent *ee = (const struct ext4_extent *)(const void *)(node + 12);
        for (uint16_t i = 0; i < eh->eh_entries; ++i) {
            uint32_t start = ee[i].ee_block;
            uint32_t len = ee[i].ee_len <= 0x8000u ? ee[i].ee_len : (uint32_t)(ee[i].ee_len - 0x8000u);
            if (logical >= start && logical < start + len) {
                uint64_t phys = ((uint64_t)ee[i].ee_start_hi << 32) | ee[i].ee_start_lo;
                *physical = (uint32_t)(phys + (logical - start));
                return 0;
            }
        }
        return 1; /* 空洞 */
    }
    /* 内部节点：二分找最后一个 ei_block <= logical 的 idx。 */
    const struct ext4_extent_idx *ix = (const struct ext4_extent_idx *)(const void *)(node + 12);
    if (!eh->eh_entries) return 1;
    int pick = -1;
    for (uint16_t i = 0; i < eh->eh_entries; ++i)
        if (ix[i].ei_block <= logical) pick = i;
        else break;
    if (pick < 0) return 1;
    uint64_t leaf = ((uint64_t)ix[pick].ei_leaf_hi << 32) | ix[pick].ei_leaf_lo;
    uint8_t block[4096];
    if (ext2_read_block((uint32_t)leaf, block) != 0) return -5;
    return ext4_extent_walk(f, block, leaf, logical, physical, depth_left - 1);
}

static int ext4_extent_map(const struct extfs_features *f,
                           const uint8_t inode_body[60], uint32_t logical,
                           uint32_t *physical)
{
    return ext4_extent_walk(f, inode_body, 0, logical, physical, 3);
}
```

- [ ] **步骤 3：实现 `ext4_extent_insert`**

语义与算法（完整实现按此写，签名固定）：

```c
/* 向 inode body 的 extent 树插入 logical 起 count 个块的映射。
 * 物理块由调用方通过 out[] 回收（分配器复用 ext2_allocate_block 逐块取号，
 * 连续号合并为一段 extent）。返回 0 成功；-28=-ENOSPC；-5=-EIO。 */
static int ext4_extent_insert(const struct extfs_features *f, uint8_t inode_body[60],
                              uint32_t logical, uint32_t count, uint32_t *out);
```

实现要点（逐条落实，不留模糊空间）：

1. 调 `ext2_allocate_block` 逐块分配 `count` 个物理块，结果写 `out[]`；把连续物理号折叠为段（`ee_len` ≤ 32768，不加 uninitialized 位）。
2. 对每一段：在 depth 0 leaf 里找 `ee_block` 相邻的既有 extent，若物理也连续则合并（扩 `ee_len` 或前移 `ee_block`），否则占用一个空槽。
3. leaf 满时分裂：`inode_body` 根升为 idx（depth 0→1）：分配新 leaf 块，把原 leaf 内容写入，根 idx 留两条目（原 leaf 的最小逻辑块 + 新 leaf）。depth 1 时新段进对应 leaf，leaf 再满则同理开新 leaf 挂到根 idx；根 idx 本身满（`eh_entries == eh_max`）再升 depth 2（两层 idx），以此类推到 depth 3，超过返回 `-28`。
4. 每次改动 leaf/idx 块后立即 `ext2_write_block` 落盘；inode body 的 `eh_entries/eh_max/eh_depth` 同步更新。
5. `eh_max` 计算：inode body 根为 4；块内 leaf/idx 为 `4096 / sizeof(struct ext4_extent)`（leaf 340）与 `4096 / sizeof(struct ext4_extent_idx)`（idx 340）。

`storage_ext2_write.c` 的 `ext2_write_new_run`（现 1 行起）入口分派：`inode->flags & 0x00080000` 时走 `ext4_extent_insert` 完成映射后再按现语义写数据块（数据初始化先于映射发布的约定保留）。

```c
/* 回放 s_start 指向的已提交事务，丢弃未提交尾部，把日志截断为干净
 * （写回 s_start=0 的日志超级块）。事务提交前崩溃 → 整体丢弃；
 * 提交后崩溃 → 全部元数据影子回放到 fs 块。revoke 块列出的 fs 块
 * 即使出现在更早描述符中也不回放（其影子已被后续删除作废）。
 * 返回 0；块写失败 -5 并将卷标记只读。 */
static int journal_recover(const struct extfs_features *f, struct journal_state *j)
```

实现要点：

1. 若 `journal_scan` 得 `pending == 0` 但 `s_start != 0`（只有半截事务）：直接跳到步骤 5 截断。
2. 重走事务环，先扫 revoke 块收集撤销表（线性数组或位图，测试规模用 `uint32_t revoked[]` 最多 1024 项，溢出按全撤销处理并记日志）。
3. 再走描述符+数据块对：对每个 tag 指向的 fs 块，若不在撤销表中，把紧随的数据块内容 `ext2_write_block(fs_block, data)` 写回；`DELETED` 标志的 tag 跳过（块已释放）。`ESCAPE` 标志的数据块先反转头部 4 字节再写（JBD2 逃生约定）。
4. 仅回放本次扫描认定的**一条**已提交事务（`pending_seq`）；序号更老的事务不应回存（s_start 恒指最新尾部）。
5. 截断：写日志超级块 `s_start = 0`、`s_sequence = pending_seq + 1`，`ext2_write_block` 落盘。
6. 任一写失败：置卷 ro（复用 `storage_state.c` 的只转 ro 状态位），返回 `-5`。

签名与语义：

```c
static int journal_begin(struct journal_state *j);      /* 开启事务；无空闲日志先 checkpoint */
static int journal_log_metadata(struct journal_state *j, uint32_t fs_block, const void *data);
                                                     /* 影子写：数据块入事务缓存，满则 flush 描述符 */

```

实现要点：

1. 影子写缓冲：每卷 `journal_state` 持 `struct { uint32_t fs_block; uint8_t data[4096]; } pending[]` 数组（上限 64 项）；满时写出一对「描述符块 + 数据块序列」到日志环（序号 = `j->sequence`），继续收尾。
2. `commit`：写 COMMIT 块（`h_sequence` 同事务），更新日志超级块 `s_start` 指向本事务描述符块、`s_sequence` 已递增预留，`ext2_write_block` 顺序保证先数据后提交（提交块是提交点）。
3. `checkpoint`：对每项影子 `ext2_write_block(fs_block, data)` 刷回原位，然后写日志超级块 `s_start = 0`。
4. 日志满判定：`s_start` 到当前写指针距离 < `pending 块数 + 2` 时先 `journal_checkpoint`。
5. 写路径接入（`storage_ext2_write.c` / `storage_ext2.c` 的元数据写入点——位图、inode 表、目录块、组描述符、超级块）：有日志卷（`g_storage.ext4_features.has_journal && journal 已加载`）时，写盘前先 `journal_log_metadata` 影子同一内容，再走原直写；操作边界（`storage_write_batch` 提交点、mkdir/unlink/rename 等系统调用路径收尾处）包 `journal_begin/commit`。数据块直写不经日志。
6. ext2 卷（无日志）完全走原路径，零行为变化。
